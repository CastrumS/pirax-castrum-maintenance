// Temporary diagnosis probe: the exact withAlteredFile() PHP mechanism, repeated, with every step's
// return value and readback recorded. Not committed.
import { afterAll, beforeAll, expect, setDefaultTimeout, test } from "bun:test";
import { startHarness, type Harness } from "./harness";

setDefaultTimeout(180_000);
let h: Harness;
beforeAll(async () => {
  h = await startHarness({ run: "fixture-probe" });
}, 600_000);
afterAll(async () => {
  await h?.stop();
}, 60_000);

test("probe: copy/file_put_contents/rename cycles across Playground workers", async () => {
  await h.php(`file_put_contents(WP_PLUGIN_DIR . '/pirax-probe.txt', "original\\n"); return true;`);
  const rows: unknown[] = [];
  for (let cycle = 0; cycle < 8; cycle++) {
    const alter = await h.php<any>(`
      $file = WP_PLUGIN_DIR . '/pirax-probe.txt';
      $source = file_get_contents($file);
      $copied = copy($file, "$file.pirax-original");
      $copyError = $copied ? null : (error_get_last()['message'] ?? null);
      $put = file_put_contents($file, "altered\\n");
      return ['tid' => getmypid(), 'source' => $source, 'copied' => $copied, 'copyError' => $copyError, 'put' => $put,
        'backupExists' => file_exists("$file.pirax-original"), 'backup' => @file_get_contents("$file.pirax-original")];
    `);
    const restore = await h.php<any>(`
      $file = WP_PLUGIN_DIR . '/pirax-probe.txt';
      $renamed = rename("$file.pirax-original", $file);
      $renameError = $renamed ? null : (error_get_last()['message'] ?? null);
      return ['tid' => getmypid(), 'renamed' => $renamed, 'renameError' => $renameError, 'after' => file_get_contents($file)];
    `);
    rows.push({ cycle, alter, restore });
  }
  console.log(JSON.stringify(rows, null, 1));
  expect(rows.length).toBe(8);
}, 300_000);
